import sys import time import urllib.request import urllib.error def archive(url): if "github.com" not in url: sys.exit("Ye GitHub URL nahi hai.") req = urllib.request.Request( "https://web.archive.org/save/" + url, headers={"User-Agent": "Mozilla/5.0"}, ) for attempt in range(3): try: with urllib.request.urlopen(req, timeout=120) as r: loc = r.headers.get("Content-Location") return "https://web.archive.org" + loc if loc else r.geturl() except urllib.error.HTTPError as e: if e.code == 429: print("Rate limit, 60 sec ruk raha hoon...") time.sleep(60) else: print(f"HTTP {e.code}, dobara try...") time.sleep(10) except Exception as e: print(f"Error: {e}, dobara try...") time.sleep(10) return None if __name__ == "__main__": if len(sys.argv) != 2: sys.exit("Use: python gh_archive.py https://github.com/user/repo") link = archive(sys.argv[1]) print("Saved:", link) if link else print("Save nahi ho paya.")
