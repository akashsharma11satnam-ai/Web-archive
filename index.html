const SAVE_URL = 'https://web.archive.org/save/'; const DELAY_MS = 10000; // har save ke beech 10 sec const MAX_RUN_MS = 5 * 60 * 1000; // Apps Script ki 6 min limit se pehle ruk jao function onOpen() { SpreadsheetApp.getUi() .createMenu('GitHub Wayback Saver') .addItem('Save pending pages', 'savePending') .addItem('Add issues/pulls/wiki rows for repo links', 'expandRepos') .addItem('Setup headers', 'setupSheet') .addToUi(); } function setupSheet() { const sh = SpreadsheetApp.getActiveSheet(); sh.getRange(1, 1, 1, 4) .setValues([['URL', 'Status', 'Snapshot link', 'Saved at']]) .setFontWeight('bold'); sh.setFrozenRows(1); sh.setColumnWidth(1, 380); sh.setColumnWidth(3, 380); } // Sirf GitHub ke hosts allow: github.com, gist.github.com, raw.githubusercontent.com, *.github.io function isGithubUrl(url) { const m = url.match(/^https?:\/\/([^\/?#:]+)/i); if (!m) return false; const host = m[1].toLowerCase(); return host === 'github.com' || host === 'www.github.com' || host === 'gist.github.com' || host === 'raw.githubusercontent.com' || /\.github\.io$/.test(host); } function normalizeUrl(raw) { let url = String(raw).trim(); if (!url) return ''; if (!/^https?:\/\//i.test(url)) url = 'https://' + url; return url; } // Repo link (github.com/user/repo) ke liye issues, pulls, releases, tags, wiki rows add karta hai function expandRepos() { const ss = SpreadsheetApp.getActive(); const sh = SpreadsheetApp.getActiveSheet(); const last = sh.getLastRow(); if (last < 2) { ss.toast('Column A mein row 2 se links daalo.', 'GitHub Wayback Saver', 6); return; } const urls = sh.getRange(2, 1, last - 1, 1).getValues().map(r => normalizeUrl(r[0])); const existing = {}; urls.forEach(u => { if (u) existing[u.replace(/\/$/, '')] = true; }); const extras = ['issues', 'pulls', 'releases', 'tags', 'wiki']; const toAdd = []; urls.forEach(u => { const m = u.match(/^https?:\/\/(?:www\.)?github\.com\/([^\/?#]+)\/([^\/?#]+)\/?$/i); if (!m) return; const base = 'https://github.com/' + m[1] + '/' + m[2]; extras.forEach(e => { const full = base + '/' + e; if (!existing[full]) { existing[full] = true; toAdd.push([full]); } }); }); if (!toAdd.length) { ss.toast('Koi naya repo page add nahi hua.', 'GitHub Wayback Saver', 6); return; } sh.getRange(sh.getLastRow() + 1, 1, toAdd.length, 1).setValues(toAdd); ss.toast(toAdd.length + ' rows add ho gayi.', 'GitHub Wayback Saver', 6); } function savePending() { const ss = SpreadsheetApp.getActive(); const sh = SpreadsheetApp.getActiveSheet(); const last = sh.getLastRow(); if (last < 2) { ss.toast('Column A mein row 2 se GitHub links daalo.', 'GitHub Wayback Saver', 6); return; } const data = sh.getRange(2, 1, last - 1, 2).getValues(); const start = Date.now(); let saved = 0, failed = 0, skipped = 0, stopReason = ''; for (let i = 0; i < data.length; i++) { const row = i + 2; const url = normalizeUrl(data[i][0]); const status = String(data[i][1]).trim(); if (!url || status === 'Saved') continue; if (!isGithubUrl(url)) { sh.getRange(row, 2).setValue('Skipped: GitHub URL nahi hai'); skipped++; continue; } if (Date.now() - start > MAX_RUN_MS) { stopReason = 'Time limit aa gayi, menu se dobara chalao.'; break; } sh.getRange(row, 2).setValue('Saving...'); SpreadsheetApp.flush(); const r = saveToWayback(url); const when = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm'); if (r.ok) { sh.getRange(row, 2, 1, 3).setValues([['Saved', r.link, when]]); saved++; } else { sh.getRange(row, 2, 1, 3).setValues([['Failed: ' + r.msg, '', when]]); failed++; if (r.rateLimited) { stopReason = 'Rate limit lag gayi. 2-3 minute ruk ke dobara chalao.'; break; } } SpreadsheetApp.flush(); Utilities.sleep(DELAY_MS); } ss.toast(saved + ' saved, ' + failed + ' failed, ' + skipped + ' skipped. ' + stopReason, 'GitHub Wayback Saver', 10); } function saveToWayback(url) { let lastMsg = ''; for (let attempt = 0; attempt < 2; attempt++) { try { const res = UrlFetchApp.fetch(SAVE_URL + url, { muteHttpExceptions: true, followRedirects: true, headers: { 'User-Agent': 'Mozilla/5.0 (Google Apps Script)' } }); const code = res.getResponseCode(); if (code === 429) return { ok: false, msg: 'rate limited', rateLimited: true }; if (code >= 200 && code < 300) { const headers = res.getHeaders(); let loc = ''; for (const k in headers) { if (k.toLowerCase() === 'content-location') loc = headers[k]; } if (loc) return { ok: true, link: 'https://web.archive.org' + loc }; const snap = latestSnapshot(url); return { ok: true, link: snap || 'https://web.archive.org/web/*/' + url }; } lastMsg = 'HTTP ' + code; if (code < 500) break; // 4xx par retry bekaar hai Utilities.sleep(10000); // 5xx par ek baar dobara } catch (e) { lastMsg = String(e).slice(0, 80); Utilities.sleep(10000); } } return { ok: false, msg: lastMsg }; } function latestSnapshot(url) { try { const res = UrlFetchApp.fetch( 'https://archive.org/wayback/available?url=' + encodeURIComponent(url), { muteHttpExceptions: true } ); const json = JSON.parse(res.getContentText()); const snap = json.archived_snapshots && json.archived_snapshots.closest; return snap && snap.url ? snap.url.replace(/^http:/, 'https:') : ''; } catch (e) { return ''; } }
